<script>
var currentLang = 'ar';
var tasbihCount = 0;

function toggleLang() {
  currentLang = currentLang === 'ar' ? 'en' : 'ar';
  document.getElementById('lang-btn').innerText = currentLang === 'ar' ? 'EN' : 'ع';
  document.documentElement.dir = currentLang === 'ar' ? 'rtl' : 'ltr';
  alert(currentLang === 'ar' ? 'تم تحويل الواجهة للعربية 🌸' : 'Interface Switched to English ✨');
}

function toggleTheme() {
  var isDark = document.documentElement.getAttribute('data-theme') === 'dark';
  document.documentElement.setAttribute('data-theme', isDark ? 'light' : 'dark');
}

function setTab(sec) {
  var sections = document.querySelectorAll('.section');
  sections.forEach(s => s.style.display = 'none');
  document.getElementById('sec-' + sec).style.display = 'block';

  var navBtns = document.querySelectorAll('nav button');
  navBtns.forEach(b => b.classList.remove('active'));
  event.currentTarget.classList.add('active');
}

function toggleCheck(btn) { btn.classList.toggle('checked'); }

function resetDay() {
  document.getElementById('p1').value = '';
  document.getElementById('p2').value = '';
  document.getElementById('p3').value = '';
  alert('تم تصفير اليوم بسلام! ابدئي الآن خطوة خفيفة جديدة 🌸');
}

function generateRoutine() {
  var val = document.getElementById('routine-time').value;
  document.getElementById('routine-result').innerText = "تم توليد: " + val;
}

function addAchievement() {
  var input = document.getElementById('achieve-in');
  if(!input.value.trim()) return;
  var list = document.getElementById('achieve-list');
  var div = document.createElement('div');
  div.className = 'item-row';
  div.innerHTML = `<span>🏆 ${input.value}</span><span class="badge badge-gold">فوز رائع</span>`;
  list.appendChild(div);
  input.value = '';
}

function countUp() {
  tasbihCount++;
  document.getElementById('counter-val').innerText = tasbihCount;
}

function resetCount() {
  tasbihCount = 0;
  document.getElementById('counter-val').innerText = 0;
}

function startTimer(min) {
  alert(`بدأت جلسة التركيز لمدة ${min} دقيقة. استمتعي بإنجازكِ بذكاء 🌸`);
}

function addVisionImage(e) {
  var file = e.target.files[0];
  if(!file) return;
  var reader = new FileReader();
  reader.onload = function(event) {
    var box = document.getElementById('vision-box');
    var div = document.createElement('div');
    div.className = 'vision-card';
    div.innerHTML = `<img src="${event.target.result}">`;
    box.appendChild(div);
  }
  reader.readAsDataURL(file);
}
