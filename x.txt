<script>
  var num = 0;

  function openPage(pageId, btn) {
    // إخفاء كافة الصفحات
    var pages = document.querySelectorAll('.page-content');
    pages.forEach(p => p.classList.remove('active'));

    // إظهار الصفحة المطلوبة
    document.getElementById(pageId).classList.add('active');

    // تحديث أزرار التنقل
    var btns = document.querySelectorAll('.chip-btn, nav button');
    btns.forEach(b => b.classList.remove('active'));
    
    if(btn) btn.classList.add('active');
  }

  function countUp() {
    num++;
    document.getElementById('counter').innerText = num;
  }

  function resetCount() {
    num = 0;
    document.getElementById('counter').innerText = 0;
  }

  function addAchieve() {
    var val = document.getElementById('achieveInput').value;
    if(!val) return;
    var list = document.getElementById('achieveList');
    var div = document.createElement('div');
    div.className = 'row';
    div.innerHTML = `<span>🏆 ${val}</span><span class="tag">فوز ممتاز</span>`;
    list.appendChild(div);
    document.getElementById('achieveInput').value = '';
  }

  function toggleLang() {
    var b = document.getElementById('langBtn');
    b.innerText = b.innerText === 'EN' ? 'ع' : 'EN';
  }

  function toggleTheme() {
    var isDark = document.body.getAttribute('data-theme') === 'dark';
    if(isDark) document.body.removeAttribute('data-theme');
    else document.body.setAttribute('data-theme', 'dark');
  }
</script>
